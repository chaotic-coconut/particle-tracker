83struct Trajectory {
84 uint64_t seed_id = 0;
85 std::vector<float> lon;
86 std::vector<float> lat;
87 std::vector<float> u;
88 std::vector<float> v;
89 int32_t first_idx = std::numeric_limits<int32_t>::max();
90
91 inline void append(float lon_r, float lat_r, float u_ms, float v_ms,
92 int32_t idx) {
93 lon.push_back(lon_r);
94 lat.push_back(lat_r);
95 u.push_back(u_ms);
96 v.push_back(v_ms);
97 if (idx < first_idx)
98 first_idx = idx;
99 }
100
101 inline void flush_to(pkd2::Writer &w_pos, pkd2::Writer &w_vel,
102 uint32_t stride = 1) {
103 if (lon.empty())
104 return;
105 std::reverse(lon.begin(), lon.end());
106 std::reverse(lat.begin(), lat.end());
107 std::reverse(u.begin(), u.end());
108 std::reverse(v.begin(), v.end());
109 // Positions
110 w_pos.add_traj_rad<float>(seed_id, lon.data(), lat.data(),
111 static_cast<uint32_t>(lon.size()), first_idx,
112 stride);
113 // Velocities
114 // lon channel = u, lat channel = v (units: m/s).
115 w_vel.add_traj_rad<float>(seed_id, u.data(), v.data(),
116 static_cast<uint32_t>(u.size()), first_idx,
117 stride);
118 lon.clear();
119 lat.clear();
120 u.clear();
121 v.clear();
122 first_idx = std::numeric_limits<int32_t>::max();
123 }
124};

155static inline int64_t snap_to_grid(int64_t t_s, int32_t dt_s) {
156 const double x = static_cast<double>(t_s) / static_cast<double>(dt_s);
157 return static_cast<int64_t>(std::llround(x)) * static_cast<int64_t>(dt_s);
158}
159
160static inline int64_t sec_to_idx_exact(int64_t t_s, int32_t dt_s) {
161 if (dt_s <= 0)
162 throw std::runtime_error("dt_s <= 0");
163 if (t_s % dt_s == 0)
164 return t_s / dt_s;
165 return static_cast<int64_t>(
166 std::llround(static_cast<double>(t_s) / static_cast<double>(dt_s)));
167}

169inline void normalize_seed_times_to_t2000(std::vector<SeedInit> &seeds,
170 SeedTimeEpoch epoch) {
171 if (epoch == SeedTimeEpoch::Since2000)
172 return;
173 for (auto &s : seeds) {
174 s.t_start_s -= UNIX_TO_2000;
175 if (s.has_stop)
176 s.t_stop_s -= UNIX_TO_2000;
177 }
178}

180// IMPORTANT: do NOT “fix” ordering by swapping only times.
181// We only snap times here.
182inline void snap_seed_times_to_grid(std::vector<SeedInit> &seeds,
183 int32_t dt_s) {
184 for (auto &s : seeds) {
185 s.t_start_s = snap_to_grid(s.t_start_s, dt_s);
186 if (s.has_stop)
187 s.t_stop_s = snap_to_grid(s.t_stop_s, dt_s);
188 }
189}

191// --------------------------------------------------------------------
192// Enqueue seeds (sets integer indices and stop target)
193// --------------------------------------------------------------------
194static inline void enqueue_seeds(const std::vector<SeedInit> &seeds,
195 int32_t dt_s) {
196 for (const auto &s : seeds) {
197 const Date d = dateFromSecondsSince2000(static_cast<TimeType>(s.t_start_s));
198 auto &vec = all_particles[d];
199
200 ParticleState ps;
201 ps.seed_id = s.seed_id;
202
203 // Start point = lon_start_rad/lat_start_rad by our record-building logic in
204 // main()
205 ps.lon = static_cast<DataType>(s.lon_start_rad);
206 ps.lat = static_cast<DataType>(s.lat_start_rad);
207
208 ps.init_idx = sec_to_idx_exact(s.t_start_s, dt_s);
209 ps.time_idx = ps.init_idx;
210
211 ps.has_stop = s.has_stop;
212 if (s.has_stop) {
213 ps.stop_lon = static_cast<DataType>(s.lon_stop_rad);
214 ps.stop_lat = static_cast<DataType>(s.lat_stop_rad);
215 ps.stop_idx = sec_to_idx_exact(s.t_stop_s, dt_s);
216 } else {
217 ps.stop_idx = 0; // unused
218 }
219
220 ps.reason = EndReason::none;
221 ps.traj.seed_id = s.seed_id;
222
223 vec.push_back(std::move(ps));
224 released_total[d]++;
225 }
226}

231static inline Date month_start_from_t2000(int64_t t2000_s) {
232 return dateFromSecondsSince2000(static_cast<TimeType>(t2000_s))
233 .beginOfMonth();
234}
235
236static inline void bucketize_seeds(const std::vector<SeedInit> &seeds,
237 SeedsByMonth &bym) {
238 bym.reserve(seeds.size() / 32 + 8);
239 for (const auto &s : seeds)
240 bym[month_start_from_t2000(s.t_start_s)].push_back(s);
241}
242
243static inline void activate_month(SeedsByMonth &bym, const Date &win_beg,
244 const Date &win_end, int32_t dt_s) {
245 const Date key = win_beg.beginOfMonth();
246 auto it = bym.find(key);
247 if (it == bym.end())
248 return;
249
250 const int64_t tmin = static_cast<int64_t>(secondsSince2000(win_beg));
251 const int64_t tmax =
252 static_cast<int64_t>(secondsSince2000(win_end.addDays(1)));
253
254 std::vector<SeedInit> window_seeds;
255 window_seeds.reserve(it->second.size());
256 for (const auto &s : it->second)
257 if (s.t_start_s >= tmin && s.t_start_s < tmax)
258 window_seeds.push_back(s);
259
260 if (!window_seeds.empty())
261 enqueue_seeds(window_seeds, dt_s);
262
263 bym.erase(it);
264}

266struct ReconGuards {
267 DataType max_current_mps = 3.;
268 int max_empty_neighbors = 10;
269 DataType step_slack = 3.;

272void propagateWindow(const Date &win_beg, const Date &win_end,
273 const MonthData &older_m, const MonthData &newer_m,
274 const std::vector<std::string> &data_vars,
275 TimeType dt_seconds, TimeType life_time_seconds,
276 pkd2::Writer &pkd_pos, pkd2::Writer &pkd_vel,
277 int32_t base_dt_s, const ReconGuards &G = ReconGuards{}) {
278 (void)win_end;
279 (void)dt_seconds;
280
281 if (data_vars.size() < 2)
282 throw std::runtime_error(
283 "propagateWindow: need at least 2 data_vars (u,v)");
284 if (!older_m.kd_sea || !newer_m.kd_sea)
285 throw std::runtime_error("propagateWindow: kd_sea missing");
286 if (base_dt_s <= 0)
287 throw std::runtime_error("propagateWindow: base_dt_s must be positive");
288
289 const TimeType dtT = static_cast<TimeType>(base_dt_s);
290
291 // Window lower bound in *index* (exact: win_beg is midnight, base_dt_s
292 // divides 86400 for 3h)
293 const int64_t t_beg_s = static_cast<int64_t>(secondsSince2000(win_beg));
294 const int64_t t_beg_idx = t_beg_s / static_cast<int64_t>(base_dt_s);
295
296 const DataType search_radius =
297 static_cast<DataType>(.08 * 1.1 / 180. * number_pi);
298 const DataType shape_param = search_radius * search_radius / (1.1 * 1.1);
299
300 auto nb_old = makeNeighborsData(older_m, search_radius, shape_param);
301 auto nb_new = makeNeighborsData(newer_m, search_radius, shape_param);
302
303 std::atomic<uint64_t> stop_life{0}, stop_no_nb{0}, stop_fast{0}, stop_step{0},
304 stop_seed0{0};
305
306 oneapi::tbb::enumerable_thread_specific<std::vector<Trajectory>> done_tls;
307
308 for (auto &bucket : all_particles) {
309 auto &vec = bucket.second;
310
311 oneapi::tbb::parallel_for(
312 oneapi::tbb::blocked_range<std::size_t>(0, vec.size(), 4096),
313 [&](const oneapi::tbb::blocked_range<std::size_t> &r) {
314 std::vector<Neighbor<DataType>> neigh;
315 neigh.reserve(16);
316
317 for (std::size_t i = r.begin(); i != r.end(); i++) {
318 auto &p = vec[i];
319 if (p.reason != EndReason::none)
320 continue;
321
322 p.traj.seed_id = p.seed_id;
323
324 int empty_nb_streak = 0;
325 int fast_streak = 0;
326
327 // Stop if we hit window bound OR stop target (seed0) OR lifetime
328 while (p.reason == EndReason::none && p.time_idx >= t_beg_idx &&
329 (!p.has_stop || p.time_idx >= p.stop_idx)) {
330 // If exactly at seed0 time, snap to seed0 position, record, and
331 // finish.
332 if (p.has_stop && p.time_idx == p.stop_idx) {
333 p.lon = p.stop_lon;
334 p.lat = p.stop_lat;
335
336 if (p.time_idx < std::numeric_limits<int32_t>::min() ||
337 p.time_idx > std::numeric_limits<int32_t>::max()) {
338 p.reason = EndReason::lifetime;
339 ++stop_life;
340 break;
341 }
342
343 p.traj.append(static_cast<float>(p.lon),
344 static_cast<float>(p.lat),
345 std::numeric_limits<float>::quiet_NaN(),
346 std::numeric_limits<float>::quiet_NaN(),
347 static_cast<int32_t>(p.time_idx));
348
349 p.reason = EndReason::lifetime; // finished
350 ++stop_seed0;
351 break;
352 }
353
354 if (p.time_idx < std::numeric_limits<int32_t>::min() ||
355 p.time_idx > std::numeric_limits<int32_t>::max()) {
356 p.reason = EndReason::lifetime;
357 ++stop_life;
358 break;
359 }
360
361 // exact lifetime in index space
362 const int64_t age_idx = (p.init_idx - p.time_idx);
363 const TimeType age_s = static_cast<TimeType>(age_idx) * dtT;
364 if (age_s >= life_time_seconds) {
365 p.reason = EndReason::lifetime;
366 ++stop_life;
367 break;
368 }
369
370 // absolute time for interpolation
371 const TimeType t_s = static_cast<TimeType>(p.time_idx) * dtT;
372
373 const Date dcur = dateFromSecondsSince2000(t_s);
374 const bool use_older = (dcur.beginOfMonth() == older_m.month);
375 const MonthData &md = use_older ? older_m : newer_m;
376 auto &nb = use_older ? nb_old : nb_new;
377
378 neigh.clear();
379 nb.computeNeighbors(neigh, p.lon, p.lat);
380
381 if (neigh.empty()) {
382 // record position, but velocity unknown
383 p.traj.append(static_cast<float>(p.lon),
384 static_cast<float>(p.lat),
385 std::numeric_limits<float>::quiet_NaN(),
386 std::numeric_limits<float>::quiet_NaN(),
387 static_cast<int32_t>(p.time_idx));
388
389 if (++empty_nb_streak > G.max_empty_neighbors) {
390 p.reason = EndReason::lifetime;
391 stop_no_nb++;
392 break;
393 }
394 --p.time_idx;
395 continue;
396 }
397 empty_nb_streak = 0;
398
399 auto vals =
400 nb.interpolateVariables(t_s, neigh, md.spl, data_vars);
401 DataType u = vals.at(data_vars[0]);
402 DataType v = vals.at(data_vars[1]);
403
404 p.traj.append(static_cast<float>(p.lon),
405 static_cast<float>(p.lat), static_cast<float>(u),
406 static_cast<float>(v),
407 static_cast<int32_t>(p.time_idx));
408
409 const double spd =
410 std::sqrt(double(u) * double(u) + double(v) * double(v));
411 if (spd > G.max_current_mps) {
412 if (++fast_streak > 4) {
413 p.reason = EndReason::lifetime;
414 ++stop_fast;
415 break;
416 }
417 } else
418 fast_streak = 0;
419
420 const DataType dt = static_cast<DataType>(base_dt_s);
421 DataType dx = -u * dt;
422 DataType dy = -v * dt;
423
424 {
425 const double max_step_km = G.max_current_mps *
426 double(base_dt_s) * G.step_slack /
427 1000.;
428 const double dr_km = std::sqrt(double(dx) * double(dx) +
429 double(dy) * double(dy)) /
430 1000.;
431 if (dr_km > max_step_km) {
432 p.reason = EndReason::lifetime;
433 ++stop_step;
434 break;
435 }
436 }
437
438 inverseTransform(dx, dy, p.lon, p.lat);
439
440 while (p.lon <= -number_pi)
441 p.lon += 2 * number_pi;
442 while (p.lon > number_pi)
443 p.lon -= 2 * number_pi;
444 if (p.lat < -number_pi / 2)
445 p.lat = -number_pi / 2;
446 if (p.lat > number_pi / 2)
447 p.lat = number_pi / 2;
448
449 --p.time_idx;
450 }
451
452 if (p.reason != EndReason::none) {
453 done_tls.local().push_back(std::move(p.traj));
454 p.traj = Trajectory{};
455 p.traj.seed_id = p.seed_id;
456 }
457 }
458 });
459 }
460
461 // flush finished trajectories serially (Writer not thread-safe)
462 for (auto &local : done_tls) {
463 for (auto &tr : local)
464 tr.flush_to(pkd_pos, pkd_vel, /*stride=*/1);
465 local.clear();
466 }
467
468 std::cerr << std::format(
469 "window {} | lifetime={} no_nb={} fast={} step={} seed0={}\n",
470 formatDate(win_beg), stop_life.load(), stop_no_nb.load(),
471 stop_fast.load(), stop_step.load(), stop_seed0.load());
472}

718 if (keep_p >= 1.0 || keep(rng))
719 seeds[w++] = std::move(seeds[i]);
720 }

724 }

734 buf[older] =
735 loadOneMonth(cur_month, level, pattern_1, pattern_2, grid_variable_names,
736 data_variable_names, dt_seconds);
737 buildKdTrees(buf[older]);
738
739 buf[newer] =

741 grid_variable_names, data_variable_names, dt_seconds);
742 buildKdTrees(buf[newer]);

744 // 24 months back (e.g. 1997-01 -> 1995-01)
745 const Date first_month = start_release.addMonths(-24).beginOfMonth();
746

748 const Date win_beg = cur_month;
749 const Date win_end = cur_month.addMonths(1).addDays(-1);
750
751 activate_month(bym, win_beg, win_end, base_dt_s);
752
753 propagateWindow(win_beg, win_end, buf[older], buf[newer],
754 data_variable_names, dt_seconds, life_limit_seconds,
755 pkd_pos, pkd_vel, base_dt_s);

757 for (auto it = all_particles.begin(); it != all_particles.end();) {
758 auto &vec = it->second;
759 vec.erase(std::remove_if(
760 vec.begin(), vec.end(),
761 [](const auto &p) { return p.reason != EndReason::none; }),
762 vec.end());
763 if (vec.empty())
764 it = all_particles.erase(it);
765 else
766 ++it;
767 }
768
769 cur_month = cur_month.addMonths(-1);
770 older ^= 1;
771 newer ^= 1;
772
773 if (cur_month > first_month) {
774 buf[newer] = MonthData{};
775 buf[newer] =
776 loadOneMonth(cur_month.addMonths(-1), level, pattern_1, pattern_2,
777 grid_variable_names, data_variable_names, dt_seconds);
778 buildKdTrees(buf[newer]);
779 }
780 }

782 // flush survivors
783 for (auto &bucket : all_particles)
784 for (auto &p : bucket.second)
785 p.traj.flush_to(pkd_pos, pkd_vel, /*stride=*/1);
786
787 pkd_pos.close();
788 pkd_vel.close();

793 std::cerr << "\nfinished computation at " << std::ctime(&end_time)
794 << "elapsed time: " << elapsed_seconds.count() << " s\n";

796 return 0;
