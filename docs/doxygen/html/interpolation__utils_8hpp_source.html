127inline constexpr double pt_two_pi = 2. * pt_pi;

137template <typename T> inline T pt_wrap_lon(T lon) {
138 // Use pt_pi constants but cast to T so comparisons behave correctly.
139 const T pi = static_cast<T>(pt_pi);
140 const T two_pi = static_cast<T>(pt_two_pi);
141
142 while (lon <= -pi)
143 lon += two_pi;
144 while (lon > pi)
145 lon -= two_pi;
146 return lon;
147}

297template <typename DataType, class DatasetAdaptor>
298struct LonLatDistanceAdaptor {
299 typedef DataType ElementType;
300 typedef DataType DistanceType;
301
302 const DatasetAdaptor &data;
303
304 explicit LonLatDistanceAdaptor(const DatasetAdaptor &data_) : data(data_) {}
305
307 inline const DataType *getPoint(std::size_t idx) const {
308 return &data.pts[2 * idx];
309 }
310
316 inline DistanceType evalMetric(const DataType *a, const DataType *b,
317 std::size_t /*size*/) const {
318 DataType dlon = a[0] - b[0];
319 if (dlon < -static_cast<DataType>(pt_pi))
320 dlon += static_cast<DataType>(pt_two_pi);
321 else if (dlon > static_cast<DataType>(pt_pi))
322 dlon -= static_cast<DataType>(pt_two_pi);
323
324 DataType dlat = a[1] - b[1];
325 DataType avgLat = (a[1] + b[1]) * static_cast<DataType>(0.5);
326
327 DataType scaled_dlon = dlon * std::cos(avgLat);
328 return scaled_dlon * scaled_dlon + dlat * dlat;
329 }
330
335 inline DistanceType evalMetric(const DataType *a, std::size_t idx_b,
336 std::size_t size) const {
337 const DataType *b = getPoint(idx_b);
338 return evalMetric(a, b, size);
339 }
340
349 template <typename T, typename IndexType>
350 inline DistanceType accum_dist(const T a, const T b, IndexType dim) const {
351 DistanceType diff = static_cast<DistanceType>(a - b);
352
353 if (dim == 0) {
354 if (diff < -static_cast<DistanceType>(pt_pi))
355 diff += static_cast<DistanceType>(pt_two_pi);
356 else if (diff > static_cast<DistanceType>(pt_pi))
357 diff -= static_cast<DistanceType>(pt_two_pi);
358 }
359
360 return diff * diff;
361 }
362};

539 std::map<std::string, DataType> interpolateVariables(
540 TimeType t, const std::vector<Neighbor<DataType>> &neighbors,
541 const std::map<std::string, std::vector<TimeSpline<DataType, TimeType>>>
542 &variable_splines,
543 const std::vector<std::string> &variable_names) {
544 std::map<std::string, DataType> out;
545 DataType den = static_cast<DataType>(0);
546
547 for (const auto &var_name : variable_names)
548 out[var_name] = static_cast<DataType>(0);
549
550 for (const auto &nbr : neighbors) {
551 for (const auto &var_name : variable_names) {
552 const auto &splines = variable_splines.at(var_name);
553
554 if (nbr.ind >= splines.size()) {
555 throw std::out_of_range("particle_tracker::NeighborsData: index " +
556 std::to_string(nbr.ind) +
557 " out of bounds for variable '" + var_name +
558 "'.");
559 }
560
561 const DataType v = splines[nbr.ind].spline(t);
562 out[var_name] += nbr.weight * v;
563 }
564 den += nbr.weight;
565 }
566
567 if (den > static_cast<DataType>(1e-8)) {
568 for (const auto &var_name : variable_names)
569 out[var_name] /= den;
570 } else {
571 for (const auto &var_name : variable_names)
572 out[var_name] = static_cast<DataType>(0);
573 }
574
575 return out;
576 }

717template <typename DataType, typename TimeType = double>
718class NeighborWithNeighborsData {
719private:
720 const KDTree<DataType> &index;
721 const std::vector<DataType> &xx_sparse{};
722 const std::vector<DataType> &yy_sparse{};
723 const DataType rad2{};
724 const DataType shape{};
725
726public:
727 NeighborWithNeighborsData(const KDTree<DataType> &idx,
728 const std::vector<DataType> &xxs,
729 const std::vector<DataType> &yys, DataType r2,
730 DataType shape_param)
731 : index(idx), xx_sparse(xxs), yy_sparse(yys), rad2(r2),
732 shape(shape_param) {}
733
741 void computeNeighbors(std::vector<NeighborWithNeighbors<DataType>> &neighbors,
742 DataType x, DataType y) {
743 x = pt_wrap_lon(x);
744
745 point<DataType> query_point = {x, y};
746
747 nf::SearchParameters searchparams;
748 searchparams.sorted = false;
749
750 std::vector<nf::ResultItem<unsigned int, DataType>> indices_dists;
751 index.radiusSearch(query_point.data(), rad2, indices_dists, searchparams);
752
753 neighbors.clear();
754 neighbors.reserve(indices_dists.size());
755
756 const DataType lon_q = pt_wrap_lon(query_point[0]);
757
758 for (const auto &result : indices_dists) {
759 NeighborWithNeighbors<DataType> nbr;
760 nbr.ind = result.first;
761
762 const DataType lon_i = pt_wrap_lon(xx_sparse.at(nbr.ind));
763 const DataType lat_i = yy_sparse.at(nbr.ind);
764
765 // True great-circle distance query -> primary neighbor (for primary
766 // weights).
767 nbr.dist = GCD(lon_q, query_point[1], lon_i, lat_i);
768 nbr.weight = std::exp(-nbr.dist * nbr.dist / shape);
769
770 // Secondary search is centered at the primary grid node.
771 point<DataType> query_point_prime = {lon_i, lat_i};
772
773 std::vector<nf::ResultItem<unsigned int, DataType>> indices_dists_prime;
774 index.radiusSearch(query_point_prime.data(), rad2, indices_dists_prime,
775 searchparams);
776
777 if (indices_dists_prime.empty()) {
778 // Loud behavior: do not keep a primary neighbor with no secondary
779 // neighborhood because derivative estimation would be undefined/noisy.
780 continue;
781 }
782
783 nbr.neighbors_prime.clear();
784 nbr.neighbors_prime.reserve(indices_dists_prime.size());
785
786 for (const auto &result_prime : indices_dists_prime) {
787 NeighborPrime<DataType> nbr_prime;
788 nbr_prime.ind_prime = result_prime.first;
789
790 if (nbr_prime.ind_prime >= xx_sparse.size() ||
791 nbr_prime.ind_prime >= yy_sparse.size()) {
792 throw std::out_of_range(
793 "particle_tracker::NeighborWithNeighborsData: index " +
794 std::to_string(nbr_prime.ind_prime) +
795 " out of bounds in computeNeighbors (secondary).");
796 }
797
798 const DataType lon_j = pt_wrap_lon(xx_sparse.at(nbr_prime.ind_prime));
799 const DataType lat_j = yy_sparse.at(nbr_prime.ind_prime);
800
801 // dist_prime = [d, ddx, ddy] from GCD_deriv implementation.
802 nbr_prime.dist_prime =
803 GCD_deriv(query_point_prime[0], query_point_prime[1], lon_j, lat_j);
804 nbr_prime.weight_prime = std::exp(-nbr_prime.dist_prime[0] *
805 nbr_prime.dist_prime[0] / shape);
806
807 nbr.neighbors_prime.push_back(nbr_prime);
808 }
809
810 neighbors.push_back(nbr);
811 }
812 }

848 std::map<std::string, VariableDerivatives<DataType>> interpolateDerivatives(
849 TimeType t, const std::vector<NeighborWithNeighbors<DataType>> &neighbors,
850 const std::map<std::string, std::vector<TimeSpline<DataType, TimeType>>>
851 &variable_splines,
852 const std::vector<std::string> &variable_names, DataType shape) const {
853 // Primary accumulators
854 std::map<std::string, DataType> value_acc;
855 std::map<std::string, DataType> dt_acc;
856 std::map<std::string, DataType> dx_acc;
857 std::map<std::string, DataType> dy_acc;
858
859 DataType den = static_cast<DataType>(0);
860
861 for (const auto &name : variable_names) {
862 value_acc[name] = static_cast<DataType>(0);
863 dt_acc[name] = static_cast<DataType>(0);
864 dx_acc[name] = static_cast<DataType>(0);
865 dy_acc[name] = static_cast<DataType>(0);
866 }
867
868 for (const auto &nbr : neighbors) {
869 const DataType w = nbr.weight;
870 den += w;
871
872 // Value + time derivative from splines at primary neighbor grid nodes
873 for (const auto &name : variable_names) {
874 const auto &splines = variable_splines.at(name);
875
876 if (nbr.ind >= splines.size()) {
877 throw std::out_of_range(
878 "particle_tracker::NeighborWithNeighborsData: index " +
879 std::to_string(nbr.ind) + " out of bounds for variable '" + name +
880 "'.");
881 }
882
883 const DataType v = splines[nbr.ind].spline(t);
884 const DataType dv = splines[nbr.ind].spline.prime(t);
885
886 value_acc[name] += w * v;
887 dt_acc[name] += w * dv;
888 }
889
890 // Per-primary-neighbor spatial derivative accumulators (heuristic)
891 DataType den_prime = static_cast<DataType>(0);
892
893 std::map<std::string, DataType> dx_prime;
894 std::map<std::string, DataType> dy_prime;
895
896 for (const auto &name : variable_names) {
897 dx_prime[name] = static_cast<DataType>(0);
898 dy_prime[name] = static_cast<DataType>(0);
899 }
900
901 for (const auto &nbr_prime : nbr.neighbors_prime) {
902 // Skip self-pairs (no geometric meaning for dv)
903 if (nbr_prime.ind_prime == nbr.ind)
904 continue;
905
906 const DataType d = nbr_prime.dist_prime[0];
907 const DataType ddx = nbr_prime.dist_prime[1];
908 const DataType ddy = nbr_prime.dist_prime[2];
909
910 // IMPORTANT: den_prime must be independent of number of variables
911 // requested.
912 den_prime += nbr_prime.weight_prime;
913
914 for (const auto &name : variable_names) {
915 const auto &splines = variable_splines.at(name);
916
917 if (nbr_prime.ind_prime >= splines.size()) {
918 throw std::out_of_range(
919 "particle_tracker::NeighborWithNeighborsData: index " +
920 std::to_string(nbr_prime.ind_prime) +
921 " out of bounds for variable '" + name + "' (secondary).");
922 }
923
924 const DataType v_i = splines[nbr.ind].spline(t);
925 const DataType v_j = splines[nbr_prime.ind_prime].spline(t);
926 const DataType dv = v_j - v_i;
927
928 // Heuristic estimator building block:
929 // weight' * (d * d(d)/dx * dv), and analogously for y.
930 dx_prime[name] += nbr_prime.weight_prime * (d * ddx * dv);
931 dy_prime[name] += nbr_prime.weight_prime * (d * ddy * dv);
932 }
933 }
934
935 if (den_prime > static_cast<DataType>(1e-16)) {
936 for (const auto &name : variable_names) {
937 dx_acc[name] += w * (dx_prime[name] / den_prime);
938 dy_acc[name] += w * (dy_prime[name] / den_prime);
939 }
940 }
941 }
942
943 std::map<std::string, VariableDerivatives<DataType>> out;
944
945 if (den > static_cast<DataType>(1e-16)) {
946 for (const auto &name : variable_names) {
947 VariableDerivatives<DataType> vd;
948
949 vd.value = value_acc[name] / den;
950 vd.time_derivative = dt_acc[name] / den;
951
952 vd.spatial_derivative_x =
953 (static_cast<DataType>(2) / shape) * (dx_acc[name] / den);
954 vd.spatial_derivative_y =
955 (static_cast<DataType>(2) / shape) * (dy_acc[name] / den);
956
957 out[name] = vd;
958 }
959 } else {
960 // No usable primary weights -> return zeros.
961 for (const auto &name : variable_names)
962 out[name] = VariableDerivatives<DataType>{};
963 }
