1// fixed_point_io_codec.hpp
2// PKD1 "record-stream" compressed helpers (zstd or gzip/zlib).
3// Requires pk_config.hpp to be included first (or compile flags defined).
4
5// NOTE ON SIZE LIMITS AND FORMAT CONSTRAINTS
6//
7// * gzip/zlib path:
8// - zlib APIs (deflate, crc32) use `uInt` for input sizes (typically 32-bit).
9// - A single gzip member therefore cannot safely process >4 GiB of
10// uncompressed data in one call.
11// - This implementation hard-fails on overflow instead of silently
12// truncating and corrupting the stream.
13// - Supporting >4 GiB would require explicit chunked deflate + rolling CRC.
14//
15// * zstd path:
16// - Zstd compression/decompression APIs use `size_t` and are not subject
17// to the 4 GiB zlib limitation.
18// - Zstd skippable frame headers store payload size as uint32_t by format;
19// this is safe here because FileHeaderPKD1 is small.
20// - Decompression accumulates the full output in memory; no hard size
21// limit is enforced for corrupted or untrusted inputs.
22

47inline void write_records_zstd(const std::string& path,
48 const std::vector<PackedParticle>& rec,
49 std::uint64_t released_count,
50 int level = 3,
51 bool append = false)
52{
53 if (!append)
54 {
55 std::ofstream os(path, std::ios::binary | std::ios::out | std::ios::trunc);
56 if (!os) throw std::runtime_error("write_records_zstd: cannot open " + path);
57
58 FileHeaderPKD1 h{};
59 h.reserved0 = released_count;
60 h.reserved1 = static_cast<uint64_t>(rec.size());
61
62 // skippable frame magic (0x184D2A50..57)
63 const uint32_t magic = 0x184D2A50u;
64 const uint32_t size = static_cast<uint32_t>(sizeof(FileHeaderPKD1));
65 os.write(reinterpret_cast<const char*>(&magic), 4);
66 os.write(reinterpret_cast<const char*>(&size), 4);
67 write_header(os, h);
68 os.flush();
69 }
70 else
71 {
72 // bump reserved1 inside skippable frame at file start
73 std::fstream hs(path, std::ios::binary | std::ios::in | std::ios::out);
74 if (!hs) throw std::runtime_error("write_records_zstd: cannot reopen for header " + path);
75
76 uint32_t magic = 0, size = 0;
77 hs.read(reinterpret_cast<char*>(&magic), 4);
78 hs.read(reinterpret_cast<char*>(&size), 4);
79 if ((magic & 0xFFFFFFF0u) != 0x184D2A50u || size < sizeof(FileHeaderPKD1))
80 throw std::runtime_error("write_records_zstd: no skippable header found");
81
82 FileHeaderPKD1 h = read_header(hs);
83 h.reserved1 += static_cast<uint64_t>(rec.size());
84
85 hs.seekp(8, std::ios::beg); // after magic+size
86 write_header(hs, h);
87 hs.flush();
88 }
89
90 if (rec.empty()) return;
91
92 std::ofstream os(path, std::ios::binary | std::ios::out | std::ios::app);
93 if (!os) throw std::runtime_error("write_records_zstd: cannot open for append " + path);
94
95 const size_t src_bytes = rec.size() * sizeof(PackedParticle);
96 const size_t bound = ZSTD_compressBound(src_bytes);
97 std::vector<char> cbuf(bound);
98
99 const size_t rc = ZSTD_compress(cbuf.data(), bound, rec.data(), src_bytes, level);
100 if (ZSTD_isError(rc))
101 throw std::runtime_error("write_records_zstd: " + std::string(ZSTD_getErrorName(rc)));
102
103 write_all(os, cbuf.data(), rc);
104}

106inline std::vector<PackedParticle> read_records_zstd(const std::string& path)
107{
108 std::ifstream is(path, std::ios::binary);
109 if (!is) throw std::runtime_error("read_records_zstd: cannot open " + path);
110
111 std::streampos start = is.tellg();
112 uint32_t magic = 0, size = 0;
113 is.read(reinterpret_cast<char*>(&magic), 4);
114 is.read(reinterpret_cast<char*>(&size), 4);
115
116 if (is && ((magic & 0xFFFFFFF0u) == 0x184D2A50u))
117 {
118 if (size < sizeof(FileHeaderPKD1))
119 throw std::runtime_error("read_records_zstd: skippable too small");
120 FileHeaderPKD1 h{};
121 read_all(is, &h);
122 if (h.magic[0] != 'P' || h.magic[1] != 'K' || h.magic[2] != 'D')
123 throw std::runtime_error("read_records_zstd: bad PKD header");
124 if (size > sizeof(FileHeaderPKD1))
125 is.seekg(static_cast<std::streamoff>(size - sizeof(FileHeaderPKD1)), std::ios::cur);
126 }
127 else
128 {
129 // backward compat: old files had raw PKD1 header
130 is.clear();
131 is.seekg(start);
132 (void)read_header(is);
133 }
134
135 ZSTD_DCtx* dctx = ZSTD_createDCtx();
136 if (!dctx) throw std::runtime_error("read_records_zstd: createDCtx failed");
137
138 const size_t in_chunk = ZSTD_DStreamInSize();
139 const size_t out_chunk = ZSTD_DStreamOutSize();
140 std::vector<char> inbuf(in_chunk), outbuf(out_chunk), raw;
141
142 ZSTD_inBuffer in { inbuf.data(), 0, 0 };
143 ZSTD_outBuffer out { outbuf.data(), outbuf.size(), 0 };
144
145 while (true)
146 {
147 if (in.pos == in.size)
148 {
149 is.read(inbuf.data(), static_cast<std::streamsize>(inbuf.size()));
150 in.size = static_cast<size_t>(is.gcount());
151 in.pos = 0;
152 if (in.size == 0) break;
153 }
154
155 out.pos = 0;
156 size_t ret = ZSTD_decompressStream(dctx, &out, &in);
157 if (ZSTD_isError(ret))
158 {
159 ZSTD_freeDCtx(dctx);
160 throw std::runtime_error("read_records_zstd: " + std::string(ZSTD_getErrorName(ret)));
161 }
162 if (out.pos) raw.insert(raw.end(), outbuf.data(), outbuf.data() + out.pos);
163 }
164
165 ZSTD_freeDCtx(dctx);
166
167 if (raw.size() % sizeof(PackedParticle) != 0)
168 throw std::runtime_error("read_records_zstd: size mismatch");
169
170 std::vector<PackedParticle> rec(raw.size() / sizeof(PackedParticle));
171 std::memcpy(rec.data(), raw.data(), raw.size());
172 return rec;
173}

182 inline uint16_t get_le16(std::istream& is)
183 {
184 unsigned char b[2]{};
185 if (!is.read(reinterpret_cast<char*>(b), 2))
186 throw std::runtime_error("get_le16: read failed");
187 return static_cast<uint16_t>(b[0]) | (static_cast<uint16_t>(b[1]) << 8);
188 }
189
190 inline uint32_t get_le32(std::istream& is)
191 {
192 unsigned char b[4]{};
193 if (!is.read(reinterpret_cast<char*>(b), 4))
194 throw std::runtime_error("get_le32: read failed");
195 return static_cast<uint32_t>(b[0])
196 | (static_cast<uint32_t>(b[1]) << 8)
197 | (static_cast<uint32_t>(b[2]) << 16)
198 | (static_cast<uint32_t>(b[3]) << 24);
199 }

201 inline void put_le16(std::ostream& os, uint16_t v)
202 {
203 unsigned char b[2] = {
204 static_cast<unsigned char>(v & 0xFFu),
205 static_cast<unsigned char>((v >> 8) & 0xFFu)
206 };
207 os.write(reinterpret_cast<const char*>(b), 2);
208 }

210 inline void put_le32(std::ostream& os, uint32_t v)
211 {
212 unsigned char b[4] = {
213 static_cast<unsigned char>(v & 0xFFu),
214 static_cast<unsigned char>((v >> 8) & 0xFFu),
215 static_cast<unsigned char>((v >> 16) & 0xFFu),
216 static_cast<unsigned char>((v >> 24) & 0xFFu)
217 };
218 os.write(reinterpret_cast<const char*>(b), 4);
219 }

221 // gzip header with FEXTRA containing FileHeaderPKD1
222 inline void write_gzip_header(std::ostream& os, const FileHeaderPKD1& h, bool with_extra)
223 {
224 const uint8_t ID1=0x1f, ID2=0x8b, CM=8;
225 uint8_t FLG = with_extra ? 0x04 : 0x00; // FEXTRA
226
227 os.put(ID1); os.put(ID2); os.put(CM); os.put(FLG);
228 put_le32(os, 0); // MTIME
229 os.put(0); // XFL
230 unsigned char os_byte=0xFF;
231 os.write(reinterpret_cast<const char*>(&os_byte), 1);
232
233 if (with_extra)
234 {
235 const uint16_t sublen = static_cast<uint16_t>(sizeof(FileHeaderPKD1));
236 const uint16_t xlen = static_cast<uint16_t>(2 + 2 + sublen);
237 put_le16(os, xlen);
238
239 os.put('P'); os.put('K'); // subfield ID
240 put_le16(os, sublen);
241 write_header(os, h);
242 }
243 }

245 inline void bump_gzip_extra_reserved1(const std::string& path, uint64_t add)
246 {
247 std::fstream hs(path, std::ios::binary | std::ios::in | std::ios::out);
248 if (!hs) throw std::runtime_error("write_records_gzip: cannot reopen " + path);
249
250 unsigned char hdr[10];
251 hs.read(reinterpret_cast<char*>(hdr), 10);
252 if (hs.gcount()!=10 || hdr[0]!=0x1f || hdr[1]!=0x8b || hdr[2]!=8)
253 throw std::runtime_error("write_records_gzip: not a gzip file");
254
255 const uint8_t flg = hdr[3];
256 if ((flg & 0x04) == 0) return; // no FEXTRA
257
258 const uint16_t xlen = gzdetail::get_le16(hs);
259 if (xlen < 4)
260 throw std::runtime_error("write_records_gzip: corrupt FEXTRA (xlen < 4)");
261
262 char si1 = 0, si2 = 0;
263 hs.get(si1); hs.get(si2);
264 const uint16_t len = gzdetail::get_le16(hs);
265
266 if (si1!='P' || si2!='K' || len < sizeof(FileHeaderPKD1))
267 throw std::runtime_error("write_records_gzip: PKD extra not found");
268
269 FileHeaderPKD1 h = read_header(hs);
270 h.reserved1 += add;
271
272 // fixed offset because layout is exactly:
273 // 10 (gzip) + 2 (XLEN) + 2 (SI) + 2 (LEN)
274 hs.seekp(10 + 2 + 2 + 2, std::ios::beg);
275 write_header(hs, h);
276 hs.flush();
277 }

280static inline void deflate_raw_to_stream(std::ostream& os,
281 const void* src,
282 std::size_t src_len,
283 int level)
284{
285 // zlib's z_stream::avail_in is uInt (usually 32-bit). Do NOT truncate silently.
286 if (src_len > static_cast<std::size_t>(std::numeric_limits<uInt>::max()))
287 throw std::runtime_error("deflate_raw_to_stream: input too large for zlib ( > 4GiB )");
288
289 z_stream zs{};
290 if (deflateInit2(&zs, level, Z_DEFLATED, -15, 9, Z_DEFAULT_STRATEGY) != Z_OK)
291 throw std::runtime_error("deflateInit2 raw failed");
292
293 zs.next_in = reinterpret_cast<Bytef*>(const_cast<void*>(src));
294 zs.avail_in = static_cast<uInt>(src_len);
295
296 unsigned char outbuf[64 * 1024];
297 int rc;
298 do
299 {
300 zs.next_out = outbuf;
301 zs.avail_out = static_cast<uInt>(sizeof(outbuf));
302
303 rc = deflate(&zs, zs.avail_in ? Z_NO_FLUSH : Z_FINISH);
304 if (rc != Z_OK && rc != Z_STREAM_END)
305 {
306 deflateEnd(&zs);
307 throw std::runtime_error("deflate error");
308 }
309
310 const std::size_t produced = sizeof(outbuf) - zs.avail_out;
311 if (produced)
312 os.write(reinterpret_cast<const char*>(outbuf),
313 static_cast<std::streamsize>(produced));
314 }
315 while (rc != Z_STREAM_END);
316
317 deflateEnd(&zs);
318}

320inline void write_records_gzip(const std::string& path,
321 const std::vector<PackedParticle>& rec,
322 std::uint64_t released_count,
323 int level = 6,
324 bool append = false)
325{
326 const std::size_t bytes = rec.size() * sizeof(PackedParticle);
327
328 // zlib crc32() length parameter is uInt -> avoid silent truncation
329 if (bytes > static_cast<std::size_t>(std::numeric_limits<uInt>::max()))
330 throw std::runtime_error("write_records_gzip: buffer too large for zlib ( > 4GiB )");
331
332 if (!append)
333 {
334 std::ofstream os(path, std::ios::binary | std::ios::out | std::ios::trunc);
335 if (!os)
336 throw std::runtime_error("write_records_gzip: cannot open " + path);
337
338 FileHeaderPKD1 h{};
339 h.reserved0 = released_count;
340 h.reserved1 = static_cast<uint64_t>(rec.size());
341
342 gzdetail::write_gzip_header(os, h, /*with_extra=*/true);
343
344 // CRC32 over uncompressed data
345 uLong crc = crc32(0L, Z_NULL, 0);
346 crc = crc32(crc,
347 reinterpret_cast<const Bytef*>(rec.data()),
348 static_cast<uInt>(bytes));
349
350 // gzip trailer ISIZE is modulo 2^32
351 const uint32_t isize32 = static_cast<uint32_t>(static_cast<uint64_t>(bytes));
352
353 // raw deflate payload
354 deflate_raw_to_stream(os, rec.data(), bytes, level);
355 if (!os.good())
356 throw std::runtime_error("write_records_gzip: ostream error during deflate");
357
358 // gzip trailer: CRC32 + ISIZE
359 gzdetail::put_le32(os, static_cast<uint32_t>(crc));
360 gzdetail::put_le32(os, isize32);
361 return;
362 }
363
364 // append path
365 if (rec.empty())
366 return;
367
368 // bump record count in first gzip member
369 gzdetail::bump_gzip_extra_reserved1(path, static_cast<uint64_t>(rec.size()));
370
371 std::ofstream os(path, std::ios::binary | std::ios::out | std::ios::app);
372 if (!os)
373 throw std::runtime_error("write_records_gzip: cannot open for append " + path);
374
375 // appended member has no extra
376 gzdetail::write_gzip_header(os, FileHeaderPKD1{}, /*with_extra=*/false);
377
378 uLong crc = crc32(0L, Z_NULL, 0);
379 crc = crc32(crc,
380 reinterpret_cast<const Bytef*>(rec.data()),
381 static_cast<uInt>(bytes));
382
383 const uint32_t isize32 = static_cast<uint32_t>(static_cast<uint64_t>(bytes));
384
385 deflate_raw_to_stream(os, rec.data(), bytes, level);
386 if (!os.good())
387 throw std::runtime_error("write_records_gzip: ostream error during deflate");
388
389 gzdetail::put_le32(os, static_cast<uint32_t>(crc));
390 gzdetail::put_le32(os, isize32);
391}

393inline std::vector<PackedParticle> read_records_gzip(const std::string& path)
394{
395 std::ifstream is(path, std::ios::binary);
396 if (!is) throw std::runtime_error("read_records_gzip: cannot open " + path);
397
398 // Rewind-safe peek for PKD extra (optional)
399 const std::streampos start = is.tellg();
400 unsigned char hdr[10]{};
401 if (!is.read(reinterpret_cast<char*>(hdr), 10))
402 throw std::runtime_error("read_records_gzip: truncated header");
403
404 if (hdr[0]==0x1f && hdr[1]==0x8b && hdr[2]==8)
405 {
406 const uint8_t flg = hdr[3];
407 if (flg & 0x04) // FEXTRA present
408 {
409 const uint16_t xlen = gzdetail::get_le16(is);
410 if (xlen < 4)
411 throw std::runtime_error("read_records_gzip: corrupt FEXTRA (xlen < 4)");
412
413 // We only inspect the *first* subfield. If it's not ours, we skip all extras.
414 char si1 = 0, si2 = 0;
415 is.get(si1); is.get(si2);
416 const uint16_t len = gzdetail::get_le16(is);
417
418 if (si1=='P' && si2=='K' && len >= sizeof(FileHeaderPKD1))
419 {
420 // Read our header payload
421 FileHeaderPKD1 h{};
422 read_all(is, &h);
423
424 if (h.magic[0] != 'P' || h.magic[1] != 'K' || h.magic[2] != 'D')
425 throw std::runtime_error("read_records_gzip: bad PKD header");
426
427 // Skip any extra bytes inside this subfield
428 const std::streamoff extra_in_subfield =
429 static_cast<std::streamoff>(len - sizeof(FileHeaderPKD1));
430 if (extra_in_subfield > 0)
431 is.seekg(extra_in_subfield, std::ios::cur);
432
433 // Skip remaining extra subfields if present
434 const uint16_t consumed = static_cast<uint16_t>(2 + 2 + len); // SI1+SI2+LEN+payload
435 if (xlen > consumed)
436 is.seekg(static_cast<std::streamoff>(xlen - consumed), std::ios::cur);
437 }
438 else
439 {
440 // Not our subfield -> skip the rest of FEXTRA.
441 // We already consumed 4 bytes (SI1,SI2,LEN). Need to skip remaining (xlen - 4).
442 is.seekg(static_cast<std::streamoff>(xlen - 4), std::ios::cur);
443 }
444 }
445 }
446
447 // rewind so zlib sees the full gzip stream from the beginning
448 is.clear();
449 is.seekg(start);
450
451 z_stream zs{};
452 if (inflateInit2(&zs, 15 + 32) != Z_OK)
453 throw std::runtime_error("inflateInit2 failed");
454
455 std::vector<char> inbuf(64*1024), outbuf(64*1024), raw;
456 zs.next_in = Z_NULL;
457 zs.avail_in = 0;
458
459 for (;;)
460 {
461 if (zs.avail_in == 0)
462 {
463 is.read(inbuf.data(), static_cast<std::streamsize>(inbuf.size()));
464 const std::streamsize got = is.gcount();
465 if (got == 0) break;
466
467 zs.next_in = reinterpret_cast<Bytef*>(inbuf.data());
468 zs.avail_in = static_cast<uInt>(got);
469 }
470
471 zs.next_out = reinterpret_cast<Bytef*>(outbuf.data());
472 zs.avail_out = static_cast<uInt>(outbuf.size());
473
474 const int ret = inflate(&zs, Z_NO_FLUSH);
475 if (ret != Z_OK && ret != Z_STREAM_END)
476 {
477 inflateEnd(&zs);
478 throw std::runtime_error("inflate error: " + std::to_string(ret));
479 }
480
481 const std::size_t produced = outbuf.size() - zs.avail_out;
482 if (produced)
483 raw.insert(raw.end(), outbuf.data(), outbuf.data() + produced);
484
485 if (ret == Z_STREAM_END)
486 {
487 // More concatenated members?
488 if (zs.avail_in > 0 || is.peek() != EOF)
489 {
490 if (inflateReset2(&zs, 15 + 32) != Z_OK)
491 {
492 inflateEnd(&zs);
493 throw std::runtime_error("inflateReset2 failed");
494 }
495 continue;
496 }
497 break;
498 }
499 }
500
501 inflateEnd(&zs);
502
503 if (raw.size() % sizeof(PackedParticle) != 0)
504 throw std::runtime_error("read_records_gzip: size mismatch");
505
506 std::vector<PackedParticle> rec(raw.size() / sizeof(PackedParticle));
507 std::memcpy(rec.data(), raw.data(), raw.size());
508 return rec;
509}
