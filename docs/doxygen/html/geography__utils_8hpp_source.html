90template<typename DataType>
91inline DataType GCD(const DataType& lon1, const DataType& lat1,
92 const DataType& lon2, const DataType& lat2)
93{
94 DataType cn =
95 std::sin(lat1) * std::sin(lat2) +
96 std::cos(lat1) * std::cos(lat2) * std::cos(lon1 - lon2);
97
98 cn = clamp(cn, static_cast<DataType>(-1), static_cast<DataType>(1));
99
100 DataType sn = std::sqrt(static_cast<DataType>(1) - cn * cn);
101
102 return std::atan2(sn, cn);
103}

135template<typename DataType>
136inline std::array<DataType, 3> GCD_deriv(const DataType& lon1, const DataType& lat1,
137 const DataType& lon2, const DataType& lat2)
138{
139 // cos(d)
140 DataType cn =
141 std::sin(lat1) * std::sin(lat2) +
142 std::cos(lat1) * std::cos(lat2) * std::cos(lon1 - lon2);
143
144 cn = clamp(cn, static_cast<DataType>(-1), static_cast<DataType>(1));
145
146 // sin(d)
147 DataType sn = std::sqrt(static_cast<DataType>(1) - cn * cn);
148
149 // d
150 DataType d = std::atan2(sn, cn);
151
152 // Guard: derivatives contain division by sin(d)
153 // Choose eps relative to DataType.
154 // - For double, 1e-14 is conservative; for float, it still works.
155 const DataType eps = static_cast<DataType>(1e-14);
156
157 if (sn <= eps)
158 {
159#if PARTICLE_TRACKER_GCD_DERIV_WARN
160#if PARTICLE_TRACKER_GCD_DERIV_WARN_ALWAYS
161 std::cerr
162 << "particle_tracker::GCD_deriv: sin(d) ~ 0 (coincident or nearly coincident points). "
163 << "Returning {d,0,0} to avoid NaN/Inf.\n";
164#else
165 // Print at most once per process.
166 static std::atomic_flag warned = ATOMIC_FLAG_INIT;
167 if (!warned.test_and_set(std::memory_order_relaxed))
168 {
169 std::cerr
170 << "particle_tracker::GCD_deriv: sin(d) ~ 0 (coincident or nearly coincident points). "
171 << "Returning {d,0,0} to avoid NaN/Inf. (This warning is printed once.)\n";
172 }
173#endif
174#endif
175 return {d, static_cast<DataType>(0), static_cast<DataType>(0)};
176 }
177
178 // Partial derivative w.r.t. lon1.
179 // Note: lon enters through cos(lon1-lon2); derivative uses sin(lon2-lon1).
180 DataType dd_dlon1 = d * std::cos(lat2) * std::sin(lon2 - lon1) / sn;
181
182 // Partial derivative w.r.t. lat1.
183 DataType dd_dlat1 =
184 d * (std::cos(lat1) * std::sin(lat2) -
185 std::sin(lat1) * std::cos(lat2) * std::cos(lon2 - lon1)) / sn;
186
187 return {d, dd_dlon1, dd_dlat1};
188}

215template<typename DataType>
216inline void inverseTransform(const DataType& x, const DataType& y, DataType& lon, DataType& lat)
217{
218 constexpr DataType EARTH_RADIUS = static_cast<DataType>(6378000.0);
219
220 DataType nrm = std::sqrt(x * x + y * y);
221
222 const DataType eps = static_cast<DataType>(1e-10);
223 if (nrm < eps)
224 return;
225
226 DataType gcd = nrm / EARTH_RADIUS;
227
228 DataType cn = std::cos(gcd);
229 DataType sn = std::sin(gcd);
230
231 DataType cl = std::cos(lat);
232 DataType sl = std::sin(lat);
233
234 DataType arg = sl * cn + (y / nrm) * cl * sn;
235 arg = clamp(arg, static_cast<DataType>(-1), static_cast<DataType>(1));
236
237 DataType lat_prime = std::asin(arg);
238
239 DataType lon_prime = lon + std::atan2(x * sn, nrm * cl * cn - y * sl * sn);
240
241 lat = lat_prime;
242 lon = lon_prime;
243}
