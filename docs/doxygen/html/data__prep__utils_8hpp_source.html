93template <typename DT>
94Grid<DT> initializeGrid(const std::string &file_path,
95 const std::vector<std::string> &variable_names) {
96 if (variable_names.size() < 3) {
97 throw std::runtime_error("initializeGrid(): variable_names must contain "
98 "lon_var, lat_var, mask_var.");
99 }
100
101 const std::string &lon_var = variable_names[0];
102 const std::string &lat_var = variable_names[1];
103 const std::string &mask_var = variable_names[2];
104
105 Grid<DT> grid;
106
107 // Create the processor for the grid initialization
108 NetCDFProcessor<DT> processor(file_path, variable_names);
109
110 // Retrieve variable shapes
111 const auto &lon_shape = processor.getVariableShape(lon_var);
112 const auto &lat_shape = processor.getVariableShape(lat_var);
113
114 // Retrieve variable data (lon/lat are treated as static and stored under key
115 // 0.0)
116 const auto &lon_data = processor.getVariableData(lon_var).at(0.0);
117 const auto &lat_data = processor.getVariableData(lat_var).at(0.0);
118
119 // Use first available time slice for mask_var (works for both time-dependent
120 // and not)
121 const auto &mask_map = processor.getVariableData(mask_var);
122 if (mask_map.empty()) {
123 throw std::runtime_error("initializeGrid(): mask variable '" + mask_var +
124 "' has no data.");
125 }
126 const auto &mask_data = mask_map.begin()->second;
127
128 // Check if 'lon' and 'lat' are 1D arrays
129 if (!(lon_shape.size() == 1 && lat_shape.size() == 1)) {
130 throw std::runtime_error("initializeGrid(): unsupported dimensions for '" +
131 lon_var + "' and '" + lat_var +
132 "'. Expected 1D lon/lat arrays.");
133 }
134
135 // Sizes of 'lon' and 'lat'
136 std::size_t n_lon = lon_shape[0];
137 std::size_t n_lat = lat_shape[0];
138
139 grid.num_points = n_lon * n_lat;
140
141 grid.lon_rad.resize(grid.num_points);
142 grid.lat_rad.resize(grid.num_points);
143 grid.in_the_sea.resize(grid.num_points);
144
145 // Convert 'lon' and 'lat' to radians
146 const DT grad_to_rad_factor = DT(number_pi / 180.);
147 std::vector<DT> lon_rad_1d(n_lon);
148 std::vector<DT> lat_rad_1d(n_lat);
149
150 for (std::size_t i = 0; i < n_lon; i++) {
151 lon_rad_1d[i] = lon_data[i] * grad_to_rad_factor;
152 }
153
154 for (std::size_t j = 0; j < n_lat; j++) {
155 lat_rad_1d[j] = lat_data[j] * grad_to_rad_factor;
156 }
157
158 // Create grid points by combining 'lon' and 'lat'
159 for (std::size_t j = 0; j < n_lat; j++) {
160 for (std::size_t i = 0; i < n_lon; i++) {
161 std::size_t idx = j * n_lon + i;
162 grid.lon_rad[idx] = lon_rad_1d[i];
163 grid.lat_rad[idx] = lat_rad_1d[j];
164 }
165 }
166
167 // Ensure that mask_data size matches grid.num_points
168 if (mask_data.size() != grid.num_points) {
169 throw std::runtime_error("initializeGrid(): mask variable '" + mask_var +
170 "' size does not match grid size.");
171 }
172
173 // Determine in_the_sea vector using mask variable (NaN = land)
174 for (std::size_t idx = 0; idx < grid.num_points; idx++) {
175 const DT m_value = mask_data[idx];
176 grid.in_the_sea[idx] = !std::isnan(m_value);
177 }
178
179 return grid;
180}

199template <typename DT, typename TT = double>
200void populateData(const std::vector<std::string> &file_paths,
201 const Grid<DT> &grid,
202 std::vector<TimeStepData<DT, TT>> &all_time_steps_data,
203 const std::vector<std::string> &data_variable_names,
204 bool check_land_points = false) {
205 (void)check_land_points;
206
207 for (const auto &file_path : file_paths) {
208 try {
209 NetCDFProcessor<DT, TT> processor(file_path, data_variable_names);
210
211 // Processing function for data population
212 auto processFunc = [&](const std::string &var_name, TT time_value,
213 const std::vector<DT> &data_vector) {
214 // Ensure data size matches grid size
215 if (data_vector.size() != grid.num_points) {
216 throw std::runtime_error("Data size (" +
217 std::to_string(data_vector.size()) +
218 ") does not match grid size (" +
219 std::to_string(grid.num_points) + ").");
220 }
221
222 // Find or create TimeStepData for this time_value
223 auto it =
224 std::find_if(all_time_steps_data.begin(), all_time_steps_data.end(),
225 [&](const TimeStepData<DT, TT> &tsd) {
226 return tsd.time_value == time_value;
227 });
228
229 if (it == all_time_steps_data.end()) {
230 // Create a new TimeStepData
231 TimeStepData<DT, TT> tsd;
232 tsd.time_value = time_value;
233
234 // Initialize data vectors for each variable
235 for (const auto &var : data_variable_names) {
236 tsd.variable_data[var].resize(grid.num_points,
237 std::numeric_limits<DT>::quiet_NaN());
238 }
239
240 all_time_steps_data.push_back(std::move(tsd));
241 it = std::prev(all_time_steps_data.end());
242 }
243
244 // Populate data for the current variable
245 auto &var_data = it->variable_data[var_name];
246
247 for (std::size_t i = 0; i < grid.num_points; i++) {
248 if (grid.in_the_sea[i]) {
249 var_data[i] = data_vector[i];
250 } else {
251 var_data[i] = std::numeric_limits<DT>::quiet_NaN();
252 }
253 }
254 };
255
256 // Process variables and populate data
257 processor.processVariables(processFunc);
258 } catch (const std::exception &e) {
259 std::cerr << "Error processing data from file '" << file_path
260 << "': " << e.what() << std::endl;
261 continue;
262 }
263 }

282 TT time_step_interval) {
283 std::vector<TT> expected_time_steps;

302template <typename DT, typename TT>
303void collectTimeSeriesData(
304 const Grid<DT> &grid, const std::vector<std::size_t> &sea_point_indices,
305 const std::vector<std::string> &variable_names,
306 const std::vector<std::string> &file_paths,
307 std::map<std::string, std::vector<std::vector<DT>>> &variable_time_series,
308 std::vector<TT> &expected_time_steps, TT time_step_interval) {
309 (void)grid;
310
311 // Determine the start and end times
312 TT earliest_time = std::numeric_limits<TT>::max();
313 TT latest_time = std::numeric_limits<TT>::lowest();
314 std::map<TT, std::string> time_to_file_map;
315
316 // Collect all available time steps from the files
317 std::set<TT> actual_time_steps_set;
318
319 for (const auto &file_path : file_paths) {
320 try {
321 NetCDFProcessor<DT, TT> processor(file_path, variable_names);
322
323 // Get time values
324 const auto &time_values = processor.getTimeValues();
325
326 for (const auto &time_step : time_values) {
327 actual_time_steps_set.insert(time_step);
328 time_to_file_map[time_step] = file_path;
329 earliest_time = std::min(earliest_time, time_step);
330 latest_time = std::max(latest_time, time_step);
331 }
332 } catch (const std::exception &e) {
333 std::cerr << "Error processing data from file '" << file_path
334 << "': " << e.what() << std::endl;
335 continue;
336 }
337 }
338
339 if (actual_time_steps_set.empty()) {
340 throw std::runtime_error("No time steps found in data files.");
341 }
342
343#ifdef DATA_PREP_DEBUG_TIME_STEPS
344 std::cerr << "\n=== Actual time steps found in data files ===\n";
345 std::cerr << "Earliest: " << earliest_time << "\n";
346 std::cerr << "Latest: " << latest_time << "\n";
347 std::cerr << "All actual time steps:\n";
348 for (auto t : actual_time_steps_set)
349 std::cerr << t << " ";
350 std::cerr << "\n\n";
351#endif
352
353 // Generate expected time steps
354 expected_time_steps =
355 generateExpectedTimeSteps(earliest_time, latest_time, time_step_interval);
356 std::size_t num_time_steps = expected_time_steps.size();
357
358#ifdef DATA_PREP_DEBUG_TIME_STEPS
359 std::cerr << "=== Expected time steps (" << num_time_steps << ") ===\n";
360#endif
361
362 // Initialize time series with NaNs
363 for (const auto &var_name : variable_names) {
364 variable_time_series[var_name].resize(sea_point_indices.size());
365 for (auto &ts_data : variable_time_series[var_name]) {
366 ts_data.resize(num_time_steps, std::numeric_limits<DT>::quiet_NaN());
367 }
368 }
369
370 // Map actual time steps to indices in expected time steps
371 std::map<TT, std::size_t> time_step_indices;
372 for (std::size_t idx = 0; idx < expected_time_steps.size(); idx++) {
373 time_step_indices[expected_time_steps[idx]] = idx;
374 }
375
376 // Load data and place it in the correct position in time series
377 for (const auto &[time_step, file_path] : time_to_file_map) {
378 try {
379 NetCDFProcessor<DT, TT> processor(file_path, variable_names);
380
381 // Load data for this time step
382 std::map<std::string, std::vector<DT>> data_at_time;
383
384 for (const auto &var_name : variable_names) {
385 const auto data_vector =
386 processor.getVariableDataAtTime(var_name, time_step);
387
388 // Collect data for sea points
389 std::vector<DT> sea_point_data;
390 sea_point_data.reserve(sea_point_indices.size());
391 for (auto idx : sea_point_indices) {
392 sea_point_data.push_back(data_vector[idx]);
393 }
394 data_at_time[var_name] = std::move(sea_point_data);
395 }
396
397 // Find the index in expected_time_steps
398 auto it = time_step_indices.find(time_step);
399 if (it != time_step_indices.end()) {
400 std::size_t t_idx = it->second;
401
402 // Place data into time series
403 for (const auto &var_name : variable_names) {
404 const auto &var_data = data_at_time.at(var_name);
405 auto &time_series = variable_time_series[var_name];
406
407 for (std::size_t i = 0; i < sea_point_indices.size(); i++) {
408 time_series[i][t_idx] = var_data[i];
409 }
410 }
411 } else {
412 std::cerr << "Warning: Time step " << time_step
413 << " not in expected time steps." << std::endl;
414 }
415 } catch (const std::exception &e) {
416 std::cerr << "Error processing data from file '" << file_path
417 << "': " << e.what() << std::endl;
418 continue;
419 }
420 }
421}

423template <typename DT, typename TT = double>
424void interpolateMissingData(std::vector<DT> &time_series,
425 const std::vector<TT> &time_points) {
426 std::size_t n = time_series.size();
427 if (n != time_points.size()) {
428 throw std::runtime_error("Time series and time points size mismatch.");
429 }
430
431 std::size_t start_idx = 0;
432 while (start_idx < n) {
433 // Skip over existing data
434 while (start_idx < n && !std::isnan(time_series[start_idx])) {
435 start_idx++;
436 }
437
438 if (start_idx == n)
439 break;
440
441 // Find the end of the NaN segment
442 std::size_t end_idx = start_idx;
443 while (end_idx < n && std::isnan(time_series[end_idx])) {
444 end_idx++;
445 }
446
447 // Handle edge cases
448 if (start_idx == 0 || end_idx == n) {
449 // Cannot interpolate, decide how to handle
450 DT fill_value;
451 if (start_idx == 0 && end_idx < n) {
452 // Missing data at the beginning, fill with first non-NaN value
453 fill_value = time_series[end_idx];
454 } else if (end_idx == n && start_idx > 0) {
455 // Missing data at the end, fill with last non-NaN value
456 fill_value = time_series[start_idx - 1];
457 } else {
458 // Entire time series is NaN
459 fill_value = static_cast<DT>(0);
460 }
461
462 for (std::size_t idx = start_idx; idx < end_idx; idx++) {
463 time_series[idx] = fill_value;
464 }
465 } else {
466 // Interpolate between existing data
467 DT y0 = time_series[start_idx - 1];
468 DT y1 = time_series[end_idx];
469 TT x0 = time_points[start_idx - 1];
470 TT x1 = time_points[end_idx];
471
472 for (std::size_t idx = start_idx; idx < end_idx; idx++) {
473 TT xi = time_points[idx];
474 DT yi = y0 + ((y1 - y0) * (xi - x0) / (x1 - x0));
475 time_series[idx] = yi;
476 }
477 }
478
479 start_idx = end_idx;
480 }

483template <typename DT, typename TT = double>
484void createSplinesForSeaPoints(
485 std::map<std::string, std::vector<std::vector<DT>>> &&variable_time_series,
486 std::map<std::string, std::vector<TimeSpline<DT, TT>>> &variable_splines,
487 TT start_time, TT time_step) {
488 for (auto &[var_name, time_series_data] : variable_time_series) {
489 std::vector<TimeSpline<DT, TT>> &splines = variable_splines[var_name];
490 splines.resize(time_series_data.size());
491
492 // each i touches a unique element -> safe to move in parallel
493 oneapi::tbb::parallel_for(
494 std::size_t(0), time_series_data.size(), [&](std::size_t i) {
495 auto &data = time_series_data[i];
496 TimeSpline<DT, TT> spline;
497 spline.values = std::move(data); // <-- move, no copy
498 spline.createSpline(start_time, time_step);
499 splines[i] = std::move(spline);
500 });
501
502 // optional: release the now-empty storage for this variable
503 time_series_data.clear();
504 time_series_data.shrink_to_fit();
505 }
506
507 // optional: free the whole (now moved-from) map early
508 variable_time_series.clear();
509 variable_time_series = {};
510}

530// -----------------------------------------------------------------------------
531// Utility: parse "YYYY-MM-DD" or "YYYY_MM_DD" into Date |
532// -----------------------------------------------------------------------------
533inline Date parseDate(const std::string &s) {
534 std::regex re(R"((\d{4})[-_](\d{2})[-_](\d{2}))");
535 std::smatch m;
536 if (!std::regex_match(s, m, re)) {
537 std::cerr << "Invalid date format: " << s << " (expected YYYY-MM-DD)\n";
538 std::exit(EXIT_FAILURE);
539 }
540 int y = std::stoi(m[1]), mth = std::stoi(m[2]), d = std::stoi(m[3]);
541 return Date(d, mth, y);
542}

556// ---------------------------------------------------------------------------
557// helper: expand pattern for one date
558// ---------------------------------------------------------------------------
559inline std::string expandPattern(const std::string &pattern, const Date &d,
560 int level) {
561 std::ostringstream lev_ss;
562 lev_ss << std::setw(4) << std::setfill('0') << level;
563 std::string out = pattern;
564
565 const std::pair<std::regex, std::string> subs[] = {
566 {std::regex("\\{YYYY\\}"), std::to_string(d.getYear())},
567 {std::regex("\\{MM\\}"), formatTwo(d.getMonth())},
568 {std::regex("\\{DD\\}"), formatTwo(d.getDay())},
569 {std::regex("\\{LEV\\}"), lev_ss.str()}};
570 for (auto const &s : subs)
571 out = std::regex_replace(out, s.first, s.second);
572
573 return out;
574}

635 }
636 for (auto &kv : splines_2)
637 kv.second.shrink_to_fit();
638 splines_2.clear(); // optional: drop map nodes
639}

679 const std::vector<std::string> &data_vars,
680 TimeType dt_seconds) {
681 // ---------- 1. daily file list for *this* month (+ padding) ------------
682 Date d_0 = month.addDays(-1); // day before 1st
683 Date d_1 = month.addMonths(1).addDays(1); // day after last
684 DateRange span(d_0, d_1);
685
686 std::vector<std::string> fp_1, fp_2;
687 buildPathVectors(span, level, pattern_1, pattern_2, fp_1, fp_2);
688
689 // ---------- 2. minimal grid init (first file of each sub-domain) -------
690 MonthData m;
691 m.month = month;
692 m.grid_1 = initializeGrid<DataType>(fp_1.front(), grid_vars);
693 m.grid_2 = initializeGrid<DataType>(fp_2.front(), grid_vars);
694
695 auto norm_pi_inplace = [](std::vector<DataType> &a) {
696 for (auto &L : a) {
697 while (L <= -number_pi)
698 L += 2 * number_pi;
699 while (L > number_pi)
700 L -= 2 * number_pi;
701 }
702 };
703
704 norm_pi_inplace(m.grid_1.lon_rad);
705 norm_pi_inplace(m.grid_2.lon_rad);
706
707 m.sea_idx_1 = getSeaPointIndices(m.grid_1);
708 m.sea_idx_2 = getSeaPointIndices(m.grid_2);
709
710 // ---------- 3. Build sea & land clouds + KD-trees for this month */
711 auto buildClouds = [&](MonthData &m) {
712 const std::size_t n_1 = m.grid_1.lon_rad.size();
713 const std::size_t n_2 = m.grid_2.lon_rad.size();
714 const std::size_t total = n_1 + n_2;
715
716 // 1) SEA CLOUD (order must match spline vectors: spl_1 then spl_2)
717 m.sea_cloud.pts.reserve((m.sea_idx_1.size() + m.sea_idx_2.size()) * 2);
718
719 // domain 1
720 for (auto idx : m.sea_idx_1) {
721 m.sea_cloud.pts.push_back(m.grid_1.lon_rad[idx]);
722 m.sea_cloud.pts.push_back(m.grid_1.lat_rad[idx]);
723 }
724 // domain 2
725 for (auto idx : m.sea_idx_2) {
726 m.sea_cloud.pts.push_back(m.grid_2.lon_rad[idx]);
727 m.sea_cloud.pts.push_back(m.grid_2.lat_rad[idx]);
728 }
729
730 m.sea_x.clear();
731 m.sea_y.clear();
732 m.sea_x.reserve(m.sea_cloud.kdtree_get_point_count());
733 m.sea_y.reserve(m.sea_cloud.kdtree_get_point_count());
734
735 for (size_t i = 0; i < m.sea_cloud.kdtree_get_point_count(); i++) {
736 m.sea_x.push_back(m.sea_cloud.kdtree_get_pt(i, 0));
737 m.sea_y.push_back(m.sea_cloud.kdtree_get_pt(i, 1));
738 }
739
740 // 2) LAND CLOUD (anything not in sea_idx_*; order doesn't matter)
741 std::vector<char> is_sea(total, 0);
742 for (auto i : m.sea_idx_1)
743 is_sea[i] = 1;
744 for (auto i : m.sea_idx_2)
745 is_sea[i + n_1] = 1;
746
747 m.land_cloud.pts.reserve((total - m.sea_idx_1.size() - m.sea_idx_2.size()) *
748 2);
749
750 for (std::size_t idx = 0; idx < total; idx++) {
751 if (is_sea[idx])
752 continue;
753
754 DataType lon, lat;
755 if (idx < n_1) {
756 lon = m.grid_1.lon_rad[idx];
757 lat = m.grid_1.lat_rad[idx];
758 } else {
759 std::size_t jdx = idx - n_1;
760 lon = m.grid_2.lon_rad[jdx];
761 lat = m.grid_2.lat_rad[jdx];
762 }
763 m.land_cloud.pts.push_back(lon);
764 m.land_cloud.pts.push_back(lat);
765 }
766
767 const auto sea_N = m.sea_cloud.kdtree_get_point_count();
768 const auto land_N = m.land_cloud.kdtree_get_point_count();
769
770 if (sea_N == 0) {
771 throw std::runtime_error("No SEA points for month " +
772 formatDate(m.month) +
773 " (check masks/variables/region).");
774 }
775 if (land_N == 0) {
776 throw std::runtime_error("No LAND points for month " +
777 formatDate(m.month) +
778 " (check masks/variables/region).");
779 }
780
781 // 3) KD trees
782 // auto kd_params=nanoflann::KDTreeSingleIndexAdaptorParams(10);
783 // m.kd_sea =std::make_unique<KDTree<DataType>>(2,m.sea_cloud ,kd_params);
784 // m.kd_land=std::make_unique<KDTree<DataType>>(2,m.land_cloud,kd_params);
785 // m.kd_sea ->buildIndex();
786 // m.kd_land->buildIndex();
787 };
788
789 buildClouds(m);
790
791 dumpCoastCSV(m, "coast_" + formatDate(m.month) + ".csv", 8);
792
793 // ---------- 3. read time series + build splines ------------------------
794 std::vector<TimeType> time_steps_1, time_steps_2;
795 // spline_map<DataType> tmp_spl_1,tmp_spl_2;
796 std::map<std::string, std::vector<std::vector<DataType>>> raw_time_series_1,
797 raw_time_series_2;
798
799 collectTimeSeriesData(m.grid_1, m.sea_idx_1, data_vars, fp_1,
800 raw_time_series_1, time_steps_1, dt_seconds);
801 collectTimeSeriesData(m.grid_2, m.sea_idx_2, data_vars, fp_2,
802 raw_time_series_2, time_steps_2, dt_seconds);
803
804 if (time_steps_1.empty() || time_steps_2.empty()) {
805 throw std::runtime_error(
806 "No time steps found for month " + formatDate(m.month) +
807 " (check file patterns / level / variable names).");
808 }
809
810 /* fill gaps, build splines */
811 auto patch = [&](auto &ts, auto &steps) {
812 for (auto &v : data_vars)
813 for (auto &vec : ts[v])
814 interpolateMissingData<DataType>(vec, steps);
815 };
816 patch(raw_time_series_1, time_steps_1);
817 patch(raw_time_series_2, time_steps_2);
818
819 createSplinesForSeaPoints(std::move(raw_time_series_1), m.spl_1,
820 static_cast<TimeType>(time_steps_1.front()),
821 dt_seconds);
822 createSplinesForSeaPoints(std::move(raw_time_series_2), m.spl_2,
823 static_cast<TimeType>(time_steps_2.front()),
824 dt_seconds);
825
826 m.spl.clear();
827 joinSplines<DataType>(std::move(m.spl_1), m.spl_2, m.spl);
828 m.spl_2.clear(); // we moved spl_1; spl_2 was copied into m.spl
829 m.spl_2 = {}; // drop its map/vector overhead
830
831 // Optional sanity check (debug builds)
832 for (auto &kv : m.spl)
833 assert(kv.second.size() == m.sea_cloud.kdtree_get_point_count());
834
835 return m;
836}
