10// ------------------------ constants ------------------------
11// Keep a single source of truth for scaling across PKD1/PKD2 and seed loaders.
12inline constexpr double fp_pi = std::numbers::pi_v<double>;
13inline constexpr uint32_t fp_ticks_per_rad = 1'000'000u; // ticks per radian
14inline constexpr double dbl_ticks_per_rad = 1'000'000.; // ticks per radian (double)
15inline constexpr uint64_t fp_ticks_per_sec = 1u; // seconds per tick (PKD1)

17// ------------------------ angle utils ------------------------
18inline double wrap_lon(double r)
19{
20 r = std::fmod(r,2.*fp_pi);
21 if (r<=-fp_pi) r+=2.*fp_pi;
22 else if (r>fp_pi) r-=2.*fp_pi;
23 return r;
24}

26// ------------------------ quantization -----------------------
27// Position: ticks = radians * fp_ticks_per_rad
28template <typename Real>
29inline int32_t quant_coord(Real rad)
30{
31 long double v = std::llround(static_cast<long double>(rad) * static_cast<long double>(fp_ticks_per_rad));
32 if (v>std::numeric_limits<int32_t>::max()) v = std::numeric_limits<int32_t>::max();
33 if (v<std::numeric_limits<int32_t>::min()) v = std::numeric_limits<int32_t>::min();
34 return static_cast<int32_t>(v);
35}
36
37template <typename Real>
38inline Real dequant_coord(int32_t q)
39{
40 return static_cast<Real>(q) / static_cast<Real>(fp_ticks_per_rad);
41}

43// Time: 1 tick = 1 second by default (PKD1 helpers)
44template <typename Real>
45inline int64_t quant_time(Real seconds)
46{
47 return static_cast<int64_t>(std::llround(static_cast<long double>(seconds)));

81// ---------------- encode/decode one record -----------------
82template<typename PosT,typename TimeT>
83inline PackedParticle encode_record(PosT lon0, PosT lat0, TimeT t0,
84 PosT lon1, PosT lat1, TimeT t1,
85 uint8_t flags = 0)
86{
87 PackedParticle r{};
88 r.lon0_q = quant_coord(wrap_lon(lon0));
89 r.lat0_q = quant_coord(lat0);
90 r.lon1_q = quant_coord(wrap_lon(lon1));
91 r.lat1_q = quant_coord(lat1);
92 r.t0_s = quant_time(t0);
93 r.t1_s = quant_time(t1);
94 r.flags = flags;
95 return r;
96}
