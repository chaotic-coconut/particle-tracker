31// Seed record:
32// - ACTIVE seed point: lon_rad/lat_rad/t_start_s (this is what simulation starts from)
33// - STOP target (optional): lon_stop_rad/lat_stop_rad/t_stop_s
34// - For convenience/back-compat: lon_start_rad/lat_start_rad mirror ACTIVE.
35struct SeedInit
36{
37 uint64_t seed_id{};
38
39 // ACTIVE point
40 double lon_rad{}; // radians in (-pi,pi]
41 double lat_rad{}; // radians
42 int64_t t_start_s{}; // seconds (t2000 for PKD2; CSV depends on caller)
43
44 // Convenience/back-compat: mirrors ACTIVE
45 double lon_start_rad{};
46 double lat_start_rad{};
47
48 // Optional STOP target
49 double lon_stop_rad{};
50 double lat_stop_rad{};
51 int64_t t_stop_s{};
52 bool has_stop{false};
53};

62 inline bool ends_with_ci(const std::string& s, const std::string& suf)
63 {
64 if (s.size() < suf.size()) return false;
65 auto lower = [](unsigned char c){ return static_cast<char>(std::tolower(c)); };
66
67 std::string a = s.substr(s.size() - suf.size());
68 std::transform(a.begin(), a.end(), a.begin(), lower);
69
70 std::string b = suf;
71 std::transform(b.begin(), b.end(), b.begin(), lower);
72
73 return a == b;
74 }

100 // Decode first+last sample of one PKD2 block.
101 // Returns two "ACTIVE-only" records: first and last (no stop metadata assigned here).
102 inline void decode_block_first_last(const pkd2::Header& hdr,
103 const pkd2::TocEntry& te,
104 FILE* f,
105 SeedInit& first_out,
106 SeedInit& last_out)
107 {
108 std::vector<unsigned char> comp(te.comp_size);
109 fseek_or_throw(f, static_cast<long>(te.file_offset), SEEK_SET, "block");
110 fread_or_throw(f, comp.data(), comp.size(), "block bytes");
111
112 auto raw = zlib_decompress_exact(comp.data(), comp.size(), te.raw_size);
113 if (raw.size() < sizeof(pkd2::BlockHeader))
114 throw std::runtime_error("PKD2: raw block too small");
115
116 const auto* bh = reinterpret_cast<const pkd2::BlockHeader*>(raw.data());
117 const uint32_t n = bh->n_points;
118 const uint16_t stride = bh->stride ? bh->stride : 1;
119 if (n == 0) throw std::runtime_error("PKD2: n_points==0 (invalid)");
120
121 // Strong sanity: pos_scale should match your writer
122 if (hdr.pos_scale_ticks_per_rad != static_cast<uint32_t>(POS_TICKS_PER_RAD))
123 throw std::runtime_error("PKD2: pos_scale_ticks_per_rad mismatch (file vs expected 1e6)");
124
125 const double inv = 1.0 / static_cast<double>(hdr.pos_scale_ticks_per_rad);
126
127 const double L0 = static_cast<double>(bh->lon0_q) * inv;
128 const double B0 = static_cast<double>(bh->lat0_q) * inv;
129
130 const int64_t t0 = hdr.t0_epoch_s
131 + static_cast<int64_t>(bh->start_index) * static_cast<int64_t>(hdr.dt_s);
132
133 const int32_t* dlon = reinterpret_cast<const int32_t*>(raw.data() + sizeof(pkd2::BlockHeader));
134 const int32_t* dlat = dlon + (n > 1 ? (n - 1) : 0);
135
136 double L = L0;
137 double B = B0;
138 for (uint32_t i = 1; i < n; ++i)
139 {
140 L += static_cast<double>(dlon[i - 1]) * inv;
141 B += static_cast<double>(dlat[i - 1]) * inv;
142 }
143
144 const int64_t t_last = t0
145 + static_cast<int64_t>(n - 1)
146 * static_cast<int64_t>(hdr.dt_s)
147 * static_cast<int64_t>(stride);
148
149 // FIRST active
150 first_out = {};
151 first_out.seed_id = bh->seed_id;
152 first_out.lon_rad = wrap_lon(L0);
153 first_out.lat_rad = B0;
154 first_out.t_start_s = t0;
155 first_out.lon_start_rad = first_out.lon_rad;
156 first_out.lat_start_rad = first_out.lat_rad;
157 first_out.has_stop = false;
158
159 // LAST active
160 last_out = {};
161 last_out.seed_id = bh->seed_id;
162 last_out.lon_rad = wrap_lon(L);
163 last_out.lat_rad = B;
164 last_out.t_start_s = t_last;
165 last_out.lon_start_rad = last_out.lon_rad;
166 last_out.lat_start_rad = last_out.lat_rad;
167 last_out.has_stop = false;
168 }

170 // PKD2 loader: attach stop target depending on which_point.
171 inline std::vector<SeedInit> load_seeds_from_pkd2_zlib(const std::string& path, SeedPoint which_point)
172 {
173 FILE* f = std::fopen(path.c_str(), "rb");
174 if (!f) throw std::runtime_error("PKD2: cannot open " + path);
175
176 pkd2::Header hdr{};
177 try
178 {
179 fread_or_throw(f, &hdr, sizeof(hdr), "header");
180
181 if (!(hdr.magic[0]=='P' && hdr.magic[1]=='K' && hdr.magic[2]=='D' && hdr.magic[3]=='2'))
182 throw std::runtime_error("PKD2: bad magic");
183 if (hdr.version != 3)
184 throw std::runtime_error("PKD2: unsupported version (expected 3)");
185 if (hdr.dt_s <= 0)
186 throw std::runtime_error("PKD2: dt_s <= 0");
187 if (hdr.n_traj == 0)
188 {
189 std::fclose(f);
190 return {};
191 }
192
193 fseek_or_throw(f, static_cast<long>(hdr.toc_offset), SEEK_SET, "toc");
194 std::vector<pkd2::TocEntry> toc(static_cast<size_t>(hdr.n_traj));
195 fread_or_throw(f, toc.data(), toc.size() * sizeof(pkd2::TocEntry), "toc bytes");
196
197 std::vector<SeedInit> out;
198 out.reserve(toc.size());
199
200 for (const auto& te : toc)
201 {
202 SeedInit first{}, last{};
203 decode_block_first_last(hdr, te, f, first, last);
204
205 // Choose ACTIVE and STOP
206 SeedInit active = (which_point == SeedPoint::First) ? first : last;
207 const SeedInit stop = (which_point == SeedPoint::First) ? last : first;
208
209 active.has_stop = true;
210 active.lon_stop_rad = stop.lon_rad;
211 active.lat_stop_rad = stop.lat_rad;
212 active.t_stop_s = stop.t_start_s;
213
214 // Mirror ACTIVE into lon_start/lat_start for back-compat
215 active.lon_start_rad = active.lon_rad;
216 active.lat_start_rad = active.lat_rad;
217
218 out.push_back(std::move(active));
219 }
220
221 std::fclose(f);
222 return out;
223 }
224 catch (...)
225 {
226 std::fclose(f);
227 throw;
228 }
229 }

231 inline std::vector<SeedInit> load_seeds_from_csv_basic(const std::string& csv_path)
232 {
233 std::ifstream is(csv_path);
234 if (!is) throw std::runtime_error("Seeds CSV: cannot open " + csv_path);
235
236 std::string header;
237 if (!std::getline(is, header))
238 throw std::runtime_error("Seeds CSV: empty file " + csv_path);
239
240 auto trim = [](std::string s){
241 while (!s.empty() && std::isspace((unsigned char)s.front())) s.erase(s.begin());
242 while (!s.empty() && std::isspace((unsigned char)s.back())) s.pop_back();
243 return s;
244 };
245
246 auto idx_of = [&](const std::string& name) -> int {
247 std::stringstream ss(header);
248 std::string tok;
249 int k = 0;
250 while (std::getline(ss, tok, ','))
251 {
252 tok = trim(tok);
253 if (tok == name) return k;
254 ++k;
255 }
256 return -1;
257 };
258
259 const int i_seed = idx_of("seed_id");
260 const int i_lon = idx_of("lon_rad");
261 const int i_lat = idx_of("lat_rad");
262 const int i_t = idx_of("t_start_s");
263 if (i_seed < 0 || i_lon < 0 || i_lat < 0 || i_t < 0)
264 throw std::runtime_error("Seeds CSV: expected columns: seed_id,lon_rad,lat_rad,t_start_s");
265
266 std::vector<SeedInit> out;
267 std::string line;
268 while (std::getline(is, line))
269 {
270 if (line.empty()) continue;
271 std::stringstream ss(line);
272 std::string f;
273 int col = 0;
274
275 SeedInit s{};
276 while (std::getline(ss, f, ','))
277 {
278 f = trim(f);
279 if (col == i_seed) s.seed_id = std::stoull(f);
280 if (col == i_lon) s.lon_rad = wrap_lon(std::stod(f));
281 if (col == i_lat) s.lat_rad = std::stod(f);
282 if (col == i_t) s.t_start_s = std::stoll(f);
283 ++col;
284 }
285
286 // CSV: only ACTIVE, no STOP.
287 s.lon_start_rad = s.lon_rad;
288 s.lat_start_rad = s.lat_rad;
289 s.has_stop = false;
290 out.push_back(s);
291 }
292 return out;
293 }
