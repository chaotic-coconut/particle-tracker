9// NOTE ON PKD2 BLOCK SIZE LIMITS
10//
11// PKD2 stores each trajectory as one independently-compressed block.
12// The TOC stores `raw_size` and `comp_size` as uint32_t, therefore:
13//
14// * Each individual trajectory block must be < 4 GiB uncompressed.
15// * Each individual trajectory block must be < 4 GiB compressed.
16//
17// This limit applies regardless of codec (zstd or zlib).
18//
19// Additionally, the zlib single-shot APIs used here (compress2/uncompress)
20// take sizes as uLong/uLongf (platform-dependent). On some platforms
21// (notably Windows), this is effectively a 4 GiB limit as well.
22// If you ever need trajectories larger than this, you must split them
23// into multiple PKD2 blocks (i.e., change the file format), or implement
24// streaming compression and store 64-bit sizes in the TOC.
25

62#pragma pack(push, 1)
63struct Header {
64 char magic[8] = {'P', 'K', 'D', '2', 0, 0, 0, 0};
65 uint32_t version = 3; // bump
66 uint8_t little = 1;
67 uint8_t _r1{0};
68 uint16_t _r2{0};
69 uint32_t pos_scale_ticks_per_rad = 1'000'000;
70 int64_t t0_epoch_s = 0;
71 int32_t dt_s = 0; // global base step (e.g., 3h)
72 uint32_t _r3{0};
73 uint64_t n_traj = 0;
74 uint64_t toc_offset = 0;
75};

77// header of the uncompressed per-trajectory block
78struct BlockHeader {
79 uint64_t seed_id;
80 int32_t start_index; // index of first stored sample on the global 3h grid
81 uint16_t stride; // >=1 (number of base steps between stored samples)
82 uint16_t _pad{0}; // keep 4-byte alignment
83 uint32_t n_points; // >=1 number of stored samples
84 int32_t lon0_q; // ticks = radians * pos_scale_ticks_per_rad
85 int32_t lat0_q; // ticks = radians * pos_scale_ticks_per_rad
86 // then (n_points-1) dlon_q[], then (n_points-1) dlat_q[] (int32 each)
87};
88
89// TOC entry at EOF
90struct TocEntry {
91 uint64_t seed_id;
92 uint64_t file_offset;
93 uint32_t comp_size;
94 uint32_t raw_size;
95 uint32_t n_points;
96 int32_t start_index;
97 uint16_t stride; // >=1
98 uint16_t _pad{0};
99};

133 // Append a trajectory given lon/lat in **radians** (the native state)
134 // Optional stride lets to decimate (e.g. stride=8 turns 3h to 24h).
135 template <class Real>
136 void add_traj_rad(uint64_t seed_id, const Real *lon_rad, const Real *lat_rad,
137 uint32_t n_points, int32_t start_index,
138 uint32_t stride = 1) {
139 static_assert(std::is_floating_point_v<Real>, "Real must be float/double");
140 if (n_points == 0)
141 return;
142
143 const bool need_copy = stride > 1 || !std::is_same_v<Real, double>;
144
145 if (need_copy) {
146 tmp_lon_.clear();
147 tmp_lat_.clear();
148 tmp_lon_.reserve((n_points + stride - 1) / stride);
149 tmp_lat_.reserve(tmp_lon_.capacity());
150 for (uint32_t i = 0; i < n_points; i += stride) {
151 tmp_lon_.push_back(static_cast<double>(lon_rad[i]));
152 tmp_lat_.push_back(static_cast<double>(lat_rad[i]));
153 }
154 emit_block(seed_id, tmp_lon_.data(), tmp_lat_.data(),
155 static_cast<uint32_t>(tmp_lon_.size()), start_index,
156 static_cast<uint16_t>(stride));
157 } else {
158 // Real == double and stride == 1
159 emit_block(seed_id, reinterpret_cast<const double *>(lon_rad),
160 reinterpret_cast<const double *>(lat_rad), n_points,
161 start_index, /*stride*/ 1);
162 }
163 }

196 static inline double clamp_lat(double r) {
197 const double hp = fp_pi * .5;
198 return (r < -hp) ? -hp : (r > hp ? hp : r);
199 }
200 static inline double unwrap_pi(double r) {
201 // (-pi,pi]
202 while (r <= -fp_pi)
203 r += 2 * fp_pi;
204 while (r > fp_pi)
205 r -= 2 * fp_pi;
206 return r;
207 }
208 static inline double unwrap_follow(double prev, double r) {
209 r = unwrap_pi(r);
210 double d = r - prev;
211 if (d > fp_pi)
212 r -= 2 * fp_pi;
213 if (d < -fp_pi)
214 r += 2 * fp_pi;
215 return r;
216 }
217 inline int32_t to_q(double radians) const {
218 long long v = llround(radians * static_cast<double>(pos_scale_));
219 if (v > std::numeric_limits<int32_t>::max())
220 v = std::numeric_limits<int32_t>::max();
221 if (v < std::numeric_limits<int32_t>::min())
222 v = std::numeric_limits<int32_t>::min();
223 return static_cast<int32_t>(v);
224 }

226 void emit_block(uint64_t seed_id, const double *lon_r, const double *lat_r,
227 uint32_t n, int32_t start_index, uint16_t stride) {
228 // build uncompressed block [BlockHeader | dlon[n-1] | dlat[n-1]]
229 BlockHeader bh{};
230 bh.seed_id = seed_id;
231 bh.start_index = start_index;
232 bh.stride = stride ? stride : 1;
233 bh.n_points = n;
234
235 double lon0 = unwrap_pi(lon_r[0]);
236 double lat0 = clamp_lat(lat_r[0]);
237 bh.lon0_q = to_q(lon0);
238 bh.lat0_q = to_q(lat0);
239
240 dlon_.assign(n > 1 ? n - 1 : 0, 0);
241 dlat_.assign(n > 1 ? n - 1 : 0, 0);
242
243 double prevL = lon0, prevB = lat0;
244 for (uint32_t i = 1; i < n; i++) {
245 double lr = unwrap_follow(prevL, lon_r[i]);
246 double br = clamp_lat(lat_r[i]);
247 dlon_[i - 1] = to_q(lr - prevL);
248 dlat_[i - 1] = to_q(br - prevB);
249 prevL = lr;
250 prevB = br;
251 }
252
253 const size_t raw_bytes = sizeof(BlockHeader) +
254 dlon_.size() * sizeof(int32_t) +
255 dlat_.size() * sizeof(int32_t);
256
257 // PKD2 format constraint: TOC stores sizes as uint32_t.
258 if (raw_bytes > static_cast<size_t>(std::numeric_limits<uint32_t>::max()))
259 throw std::runtime_error("pkd2: raw block exceeds 4GiB (format limit)");
260
261 raw_.resize(raw_bytes);
262 std::memcpy(raw_.data(), &bh, sizeof(bh));
263 if (!dlon_.empty()) {
264 std::memcpy(raw_.data() + sizeof(bh), dlon_.data(),
265 dlon_.size() * sizeof(int32_t));
266
267 std::memcpy(raw_.data() + sizeof(bh) + dlon_.size() * sizeof(int32_t),
268 dlat_.data(), dlat_.size() * sizeof(int32_t));
269 }
270
271 // compress to memory
272 uint32_t comp_sz = 0;
273
274#ifdef PK_IO_USE_ZSTD
275 {
276 const size_t bound = ZSTD_compressBound(raw_bytes);
277
278 // Still must fit PKD2's uint32_t TOC fields.
279 if (bound > static_cast<size_t>(std::numeric_limits<uint32_t>::max()))
280 throw std::runtime_error(
281 "pkd2: zstd compressBound exceeds 4GiB (format limit)");
282
283 comp_.resize(bound);
284
285 const size_t z =
286 ZSTD_compress(comp_.data(), bound, raw_.data(), raw_bytes, lvl_);
287 if (ZSTD_isError(z))
288 throw std::runtime_error(std::string("ZSTD_compress: ") +
289 ZSTD_getErrorName(z));
290
291 if (z > static_cast<size_t>(std::numeric_limits<uint32_t>::max()))
292 throw std::runtime_error(
293 "pkd2: compressed block exceeds 4GiB (format limit)");
294
295 comp_sz = static_cast<uint32_t>(z);
296 }
297#elif defined(PK_IO_USE_ZLIB)
298 {
299 // zlib APIs here take uLong/uLongf (platform-dependent width).
300 if (raw_bytes > static_cast<size_t>(std::numeric_limits<uLong>::max()))
301 throw std::runtime_error("pkd2: raw block too large for zlib uLong");
302
303 const uLongf bound = compressBound(static_cast<uLong>(raw_bytes));
304
305 if (static_cast<size_t>(bound) >
306 static_cast<size_t>(std::numeric_limits<uint32_t>::max()))
307 throw std::runtime_error(
308 "pkd2: zlib compressBound exceeds 4GiB (format limit)");
309
310 comp_.resize(static_cast<size_t>(bound));
311
312 uLongf z = bound;
313 const int rc = compress2(comp_.data(), &z,
314 reinterpret_cast<const Bytef *>(raw_.data()),
315 static_cast<uLong>(raw_bytes), lvl_);
316 if (rc != Z_OK)
317 throw std::runtime_error("pkd2: compress2 failed");
318
319 if (static_cast<size_t>(z) >
320 static_cast<size_t>(std::numeric_limits<uint32_t>::max()))
321 throw std::runtime_error(
322 "pkd2: compressed block exceeds 4GiB (format limit)");
323
324 comp_sz = static_cast<uint32_t>(z);
325 }
326#endif
327
328 // write to file, remember offset
329 long off = std::ftell(f_);
330 if (off < 0)
331 throw std::runtime_error("pkd2: ftell failed");
332
333 const size_t nw = std::fwrite(comp_.data(), 1, comp_sz, f_);
334 if (nw != comp_sz)
335 throw std::runtime_error("pkd2: fwrite failed");
336
337 TocEntry te{};
338 te.seed_id = seed_id;
339 te.file_offset = static_cast<uint64_t>(off);
340 te.comp_size = comp_sz;
341 te.raw_size = static_cast<uint32_t>(raw_bytes);
342 te.n_points = n;
343 te.start_index = start_index;
344 te.stride = bh.stride;
345
346 toc_.push_back(te);
347 }
348
349 void write_at(long off, const void *p, size_t n) {
350 if (std::fseek(f_, off, SEEK_SET) != 0)
351 throw std::runtime_error("fseek failed");
352 size_t nw = std::fwrite(p, 1, n, f_);
353 if (nw != n)
354 throw std::runtime_error("header write failed");
355 std::fseek(f_, 0, SEEK_END);
356 }

359/* ---------------------------- reader ------------------------------- */
360class Reader {
361public:
362 explicit Reader(const std::string &path) {
363 f_ = std::fopen(path.c_str(), "rb");
364 if (!f_)
365 throw std::runtime_error("pkd2::Reader: cannot open " + path);
366 read_at(0, &hdr_, sizeof(hdr_));
367 if (hdr_.magic[0] != 'P' || hdr_.magic[1] != 'K' || hdr_.magic[2] != 'D' ||
368 hdr_.magic[3] != '2')
369 throw std::runtime_error("pkd2: bad magic");
370 if (hdr_.version != 3)
371 throw std::runtime_error("pkd2: version != 3");
372 // read TOC
373 toc_.resize(hdr_.n_traj);
374 read_at(static_cast<long>(hdr_.toc_offset), toc_.data(),
375 toc_.size() * sizeof(TocEntry));
376 for (size_t i = 0; i < toc_.size(); ++i)
377 index_[toc_[i].seed_id] = static_cast<uint32_t>(i);
378 }
379 ~Reader() {
380 if (f_)
381 std::fclose(f_);
382 }

384 bool has(uint64_t seed_id) const {
385 return index_.find(seed_id) != index_.end();
386 }

388 // Read one trajectory; outputs in **radians** (to match sim)
389 void read_traj(uint64_t seed_id, std::vector<double> &lon_r,
390 std::vector<double> &lat_r, std::vector<int64_t> &t_s) const {
391 auto it = index_.find(seed_id);
392 if (it == index_.end())
393 throw std::runtime_error("pkd2: seed_id not found");
394 const TocEntry &te = toc_[it->second];
395
396 // read compressed block
397 comp_.resize(te.comp_size);
398 read_at(static_cast<long>(te.file_offset), comp_.data(), comp_.size());
399
400 // decompress
401 raw_.resize(te.raw_size);
402#ifdef PK_IO_USE_ZSTD
403 size_t out =
404 ZSTD_decompress(raw_.data(), raw_.size(), comp_.data(), comp_.size());
405 if (ZSTD_isError(out) || out != raw_.size())
406 throw std::runtime_error("ZSTD_decompress failed");
407#elif defined(PK_IO_USE_ZLIB)
408 uLongf out = static_cast<uLongf>(raw_.size());
409 int rc = uncompress(raw_.data(), &out, comp_.data(),
410 static_cast<uLong>(comp_.size()));
411 if (rc != Z_OK || out != raw_.size())
412 throw std::runtime_error("uncompress failed");
413#endif
414 // parse block
415 if (raw_.size() < sizeof(BlockHeader))
416 throw std::runtime_error("pkd2: raw too small");
417 const BlockHeader *bh = reinterpret_cast<const BlockHeader *>(raw_.data());
418 const uint16_t s = bh->stride ? bh->stride : 1;
419 if (bh->n_points != te.n_points)
420 throw std::runtime_error("pkd2: n_points mismatch");
421
422 const int32_t *dlon =
423 reinterpret_cast<const int32_t *>(raw_.data() + sizeof(BlockHeader));
424 const int32_t *dlat = dlon + (bh->n_points > 0 ? bh->n_points - 1 : 0);
425
426 lon_r.resize(bh->n_points);
427 lat_r.resize(bh->n_points);
428 t_s.resize(bh->n_points);
429
430 // reconstruct
431 const double inv_scale =
432 1. / static_cast<double>(hdr_.pos_scale_ticks_per_rad);
433 double L = static_cast<double>(bh->lon0_q) * inv_scale;
434 double B = static_cast<double>(bh->lat0_q) * inv_scale;
435 lon_r[0] = L;
436 lat_r[0] = B;
437 t_s[0] =
438 hdr_.t0_epoch_s + static_cast<int64_t>(bh->start_index) * hdr_.dt_s;
439
440 for (uint32_t i = 1; i < bh->n_points; i++) {
441 L += static_cast<double>(dlon[i - 1]) * inv_scale;
442 B += static_cast<double>(dlat[i - 1]) * inv_scale;
443 lon_r[i] = L;
444 lat_r[i] = B;
445 t_s[i] = t_s[0] + static_cast<int64_t>(i) * hdr_.dt_s * s;
446 }
447 }
