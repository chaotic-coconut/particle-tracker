\begin{DoxyItemize}
\item For each primary neighbor i:\+
\begin{DoxyItemize}
\item For each secondary neighbor j:\+
\begin{DoxyItemize}
\item dv = v\+\_\+j -\/ v\+\_\+i
\item accumulate using weight\+\_\+prime \texorpdfstring{$\ast$}{*} (d \texorpdfstring{$\ast$}{*} ddx \texorpdfstring{$\ast$}{*} dv) and (d \texorpdfstring{$\ast$}{*} ddy \texorpdfstring{$\ast$}{*} dv)
\end{DoxyItemize}
\end{DoxyItemize}
\item Normalize per-\/primary-\/neighbor by sum of weight\+\_\+prime (den\+\_\+prime)
\item Average over primary neighbors with primary weights
\item Apply final scaling factor (2/\+shape)
